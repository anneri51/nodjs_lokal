    <script>
        let allData = []; // Store all data for searching
        let currentBelege = []; // Store current belege data
        let currentBelegId = null; // Store current selected beleg ID
        let currentPositions = []; // Store current positions data
        let belegeCounts = {}; // Store belege counts for each page

        document.addEventListener('DOMContentLoaded', function() {
            fetch('http://localhost:5000/ordner_pages')
                .then(response => {
                    if (!response.ok) {
                        throw new Error('Network response was not ok');
                    }
                    return response.json();
                })
                .then(data => {
                    allData = data;
                    // First load all data, then fetch belege counts
                    displayData(allData);
                    setupCollapsible();
                    setupSearch();
                    setupPageItemClick();
                    setupBelegSearch();
                    
                    // Now fetch belege counts for each page
                    fetchBelegeCounts();
                })
                .catch(error => {
                    document.getElementById('content').innerHTML = 
                        `<div class="error">Fehler beim Laden der Daten: ${error.message}</div>`;
                });
        });

        // Function to fetch belege counts for each page
        function fetchBelegeCounts() {
            // Create an array of promises for each page
            const fetchPromises = allData.map(page => 
                fetch(`http://localhost:5000/belege_count/${page.PK_ABL_ORDNER_PAGE}`)
                    .then(response => {
                        if (!response.ok) {
                            throw new Error(`Failed to fetch count for page ${page.PK_ABL_ORDNER_PAGE}`);
                        }
                        return response.json();
                    })
                    .then(countData => {
                        // Store the count in our belegeCounts object
                        belegeCounts[page.PK_ABL_ORDNER_PAGE] = countData.count || 0;
                    })
                    .catch(error => {
                        console.error(`Error fetching count for page ${page.PK_ABL_ORDNER_PAGE}:`, error);
                        belegeCounts[page.PK_ABL_ORDNER_PAGE] = 0;
                    })
            );
            
            // Wait for all promises to resolve
            Promise.all(fetchPromises)
                .then(() => {
                    // Update the display with the counts
                    updateBelegeCountsDisplay();
                })
                .catch(error => {
                    console.error('Error fetching belege counts:', error);
                });
        }

        // Function to update the display with belege counts
        function updateBelegeCountsDisplay() {
            // Update each page item with its belege count
            document.querySelectorAll('.page-item').forEach(item => {
                const pageId = item.getAttribute('data-id');
                const count = belegeCounts[pageId] || 0;
                
                // Find or create the count element
                let countElement = item.querySelector('.belege-count');
                if (!countElement) {
                    countElement = document.createElement('span');
                    countElement.className = 'belege-count';
                    item.querySelector('div').appendChild(countElement);
                }
                
                countElement.textContent = count;
            });
        }

        function displayData(data) {
            if (!data || data.length === 0) {
                document.getElementById('content').innerHTML = 
                    '<div class="loading">Keine Daten verfügbar.</div>';
                return;
            }

            // Group data by JAHR and then by ORDNER_NAME
            const groupedData = {};
            
            data.forEach(item => {
                if (!groupedData[item.JAHR]) {
                    groupedData[item.JAHR] = {};
                }
                if (!groupedData[item.JAHR][item.ORDNER_NAME]) {
                    groupedData[item.JAHR][item.ORDNER_NAME] = [];
                }
                groupedData[item.JAHR][item.ORDNER_NAME].push(item);
            });

            // Generate HTML
            let html = '';
            
            // Sort years in descending order
            const years = Object.keys(groupedData).sort((a, b) => b - a);
            
            years.forEach(jahr => {
                html += `<div class="jahr-group">`;
                html += `<div class="jahr-title collapsed">Jahr: ${jahr}</div>`;
                html += `<div class="jahr-content" style="display:none;">`;
                
                // Sort folders alphabetically
                const ordners = Object.keys(groupedData[jahr]).sort();
                
                ordners.forEach(ordner => {
                    const pageCount = groupedData[jahr][ordner].length;
                    html += `<div class="ordner-group">`;
                    html += `<div class="ordner-title collapsed">Ordner: ${ordner} <span class="folder-count">(${pageCount})</span></div>`;
                    html += `<div class="page-items" style="display:none;">`;
                    
                    // Add each page item
                    groupedData[jahr][ordner].forEach(page => {
                        html += `<div class="page-item" data-id="${page.PK_ABL_ORDNER_PAGE}" style="${page.p_datum_belege_all_ok ? 'color: green;' : ''}">
                            <div>
                                ${page.PK_ABL_ORDNER_PAGE}: ${page.ORDNER_PAGE} (${page.DESCR}) (${page.PK_ABL_ORDNER}) - 
                                ${page.p_datum_belege_all_ok ? '✓ ' + formatDate(page.p_datum_belege_all_ok) : ''}
                                <span class="belege-count">0</span>
                            </div>
                            <div class="page-item-buttons">
                                <button class="page-item-button" onclick="showBelegeDrawer(${page.PK_ABL_ORDNER_PAGE}, event)">Belege</button>
                                <button class="page-item-button pictures" onclick="showPicturesDrawer(${page.PK_ABL_ORDNER_PAGE}, event)">Bilder</button>
                            </div>
                        </div>`;
                    });
                    
                    html += `</div></div>`;
                });
                
                html += `</div></div>`;
            });

            document.getElementById('content').innerHTML = html;
        }

        function formatDate(dateInput) {
            if (!dateInput) return '';
            
            const date = new Date(dateInput);
            if (isNaN(date.getTime())) return dateInput; // Fallback if invalid date
            
            const day = String(date.getDate()).padStart(2, '0');
            const month = String(date.getMonth() + 1).padStart(2, '0');
            const year = date.getFullYear();
            
            return `${day}.${month}.${year}`;
        }

        function setupCollapsible() {
            // Set up click handlers for year titles
            document.querySelectorAll('.jahr-title').forEach(title => {
                title.addEventListener('click', function() {
                    this.classList.toggle('collapsed');
                    const content = this.nextElementSibling;
                    if (content.style.display === 'none') {
                        content.style.display = 'block';
                    } else {
                        content.style.display = 'none';
                    }
                });
            });

            // Set up click handlers for folder titles
            document.querySelectorAll('.ordner-title').forEach(title => {
                title.addEventListener('click', function() {
                    this.classList.toggle('collapsed');
                    const content = this.nextElementSibling;
                    if (content.style.display === 'none') {
                        content.style.display = 'block';
                    } else {
                        content.style.display = 'none';
                    }
                });
            });
        }

        function setupSearch() {
            const searchInput = document.getElementById('searchInput');
            searchInput.addEventListener('input', function() {
                const searchTerm = this.value.toLowerCase();
                if (searchTerm.length === 0) {
                    displayData(allData);
                    setupCollapsible();
                    setupPageItemClick();
                    // Update counts after displaying data
                    setTimeout(updateBelegeCountsDisplay, 100);
                    return;
                }

                const filteredData = allData.filter(item => 
                    item.JAHR.toString().toLowerCase().includes(searchTerm) ||
                    item.ORDNER_NAME.toLowerCase().includes(searchTerm) ||
                    (item.ORDNER_PAGE && item.ORDNER_PAGE.toString().toLowerCase().includes(searchTerm))
                );

                displayData(filteredData);
                setupCollapsible();
                setupPageItemClick();
                
                // Update counts after displaying filtered data
                setTimeout(updateBelegeCountsDisplay, 100);
                
                // Auto-expand all groups when searching
                if (searchTerm.length > 0) {
                    document.querySelectorAll('.jahr-title.collapsed').forEach(el => el.click());
                    document.querySelectorAll('.ordner-title.collapsed').forEach(el => el.click());
                }
            });
        }

        function setupBelegSearch() {
            const belegSearchInput = document.getElementById('belegSearchInput');
            belegSearchInput.addEventListener('input', function() {
                const searchTerm = this.value.toLowerCase();
                
                if (searchTerm.length === 0) {
                    // Reset to original display if search is empty
                    displayBelege(currentBelege);
                    if (currentBelegId) {
                        loadBelegPositions(currentBelegId);
                    }
                    return;
                }
                
                // Search in belege
                const filteredBelege = currentBelege.filter(beleg => {
                    return Object.values(beleg).some(val => 
                        val !== null && val.toString().toLowerCase().includes(searchTerm)
                    );
                });
                
                // Update filtered count
                document.getElementById('filteredBelegCount').textContent = filteredBelege.length;
                
                // Highlight matches in the table
                displayFilteredBelege(filteredBelege, searchTerm);
                
                // Search in positions if they are loaded
                if (currentPositions.length > 0) {
                    const filteredPositions = currentPositions.filter(pos => {
                        return Object.values(pos).some(val => 
                            val !== null && val.toString().toLowerCase().includes(searchTerm)
                        );
                    });
                    
                    displayFilteredPositions(filteredPositions, searchTerm);
                }
            });
        }

        function displayFilteredBelege(filteredBelege, searchTerm) {
            if (filteredBelege.length === 0) {
                document.getElementById('belegGroups').innerHTML = 
                    '<div class="no-selection">Keine Belege gefunden, die dem Suchbegriff entsprechen</div>';
                document.querySelector('.positions-table').innerHTML = 
                    '<div class="no-selection">Keine Belege ausgewählt</div>';
                return;
            }
            
            // Group by BELEGDATUM
            const groupedByDate = {};
            filteredBelege.forEach(beleg => {
                const date = beleg.BELEGDATUM || 'Ohne Datum';
                if (!groupedByDate[date]) {
                    groupedByDate[date] = [];
                }
                groupedByDate[date].push(beleg);
            });
            
            // Sort dates (newest first)
            const sortedDates = Object.keys(groupedByDate).sort((a, b) => {
                if (a === 'Ohne Datum') return 1;
                if (b === 'Ohne Datum') return -1;
                return new Date(b) - new Date(a);
            });
            
            let html = '';
            
            sortedDates.forEach(date => {
                const belegeForDate = groupedByDate[date];
                
                html += `<div class="date-group">`;
                html += `<div class="date-title">${date} <span class="folder-count">(${belegeForDate.length})</span></div>`;
                html += `<div class="beleg-items">`;
                
                html += `<table><thead><tr>`;
                // Get all unique keys from the first beleg (assuming all have same structure)
                const keys = Object.keys(belegeForDate[0]).filter(key => key !== 'BELEGDATUM');
                html += keys.map(key => `<th>${key}</th>`).join('');
                html += `</tr></thead><tbody>`;
                
                belegeForDate.forEach(beleg => {
                    html += `<tr data-id="${beleg.ID}" onclick="loadBelegPositions(${beleg.PK_INP_BELEGE_ALL})" style="cursor:pointer;">`;
                    html += keys.map(key => {
                        const val = beleg[key];
                        if (val === null) return '<td></td>';
                        const strVal = val.toString();
                        if (searchTerm && strVal.toLowerCase().includes(searchTerm)) {
                            const highlighted = strVal.replace(
                                new RegExp(searchTerm, 'gi'),
                                match => `<span class="highlight">${match}</span>`
                            );
                            return `<td>${highlighted}</td>`;
                        }
                        return `<td>${strVal}</td>`;
                    }).join('');
                    html += `</tr>`;
                });
                
                html += `</tbody></table></div></div>`;
            });
            
            document.getElementById('belegGroups').innerHTML = html;
            
            // Set up click handlers for date groups
            document.querySelectorAll('.date-title').forEach(title => {
                title.addEventListener('click', function() {
                    this.classList.toggle('collapsed');
                    const content = this.nextElementSibling;
                    if (content.style.display === 'none') {
                        content.style.display = 'block';
                    } else {
                        content.style.display = 'none';
                    }
                });
            });
        }

        function displayFilteredPositions(filteredPositions, searchTerm) {
            if (filteredPositions.length === 0) {
                document.querySelector('.positions-table').innerHTML = 
                    '<div class="no-selection">Keine Positionen gefunden, die dem Suchbegriff entsprechen</div>';
                return;
            }
            
            let tableHtml = `
                <table>
                    <thead>
                        <tr>
                            ${Object.keys(filteredPositions[0]).map(key => `<th>${key}</th>`).join('')}
                        </tr>
                    </thead>
                    <tbody>
                        ${filteredPositions.map(pos => `
                            <tr>
                                ${Object.entries(pos).map(([key, val]) => {
                                    if (val === null) return '<td></td>';
                                    const strVal = val.toString();
                                    if (searchTerm && strVal.toLowerCase().includes(searchTerm)) {
                                        const highlighted = strVal.replace(
                                            new RegExp(searchTerm, 'gi'),
                                            match => `<span class="highlight">${match}</span>`
                                        );
                                        return `<td>${highlighted}</td>`;
                                    }
                                        return `<td>${strVal}</td>`;
                                }).join('')}
                            </tr>
                        `).join('')}
                    </tbody>
                </table>
            `;
            
            document.querySelector('.positions-table').innerHTML = tableHtml;
        }

        function setupPageItemClick() {
            document.querySelectorAll('.page-item').forEach(item => {
                item.addEventListener('click', function() {
                    // Remove active class from all items
                    document.querySelectorAll('.page-item').forEach(i => {
                        i.classList.remove('active');
                    });
                    
                    // Add active class to clicked item
                    this.classList.add('active');
                    
                    // Get the ID of the clicked item
                    const pageId = this.getAttribute('data-id');
                    
                    // Load belege for this page
                    loadBelege(pageId);
                });
            });
        }

        function loadBelege(pageId) {
            // Reset search
            document.getElementById('belegSearchInput').value = '';
            
            fetch(`http://localhost:5000/belege/${pageId}`)
                .then(response => {
                    if (!response.ok) {
                        throw new Error('Belege konnten nicht geladen werden');
                    }
                    return response.json();
                })
                .then(data => {
                    currentBelege = data;
                    currentPositions = [];
                    displayBelege(data);
                })
                .catch(error => {
                    document.getElementById('belegDetails').style.display = 'none';
                    document.querySelector('#rightPanel .no-selection').style.display = 'block';
                    document.querySelector('#rightPanel .no-selection').textContent = 
                        `Fehler beim Laden der Belege: ${error.message}`;
                });
        }

        function displayBelege(belege) {
            const rightPanel = document.querySelector('.right-panel-content');
            const noSelection = rightPanel.querySelector('.no-selection');
            const belegDetails = document.getElementById('belegDetails');
            
            if (belege.length === 0) {
                noSelection.textContent = 'Keine Belege für diese Seite gefunden';
                noSelection.style.display = 'block';
                belegDetails.style.display = 'none';
                document.getElementById('belegCount').textContent = '0';
                document.getElementById('filteredBelegCount').textContent = '0';
                return;
            }
            
            noSelection.style.display = 'none';
            belegDetails.style.display = 'block';
            
            // Update counts
            document.getElementById('belegCount').textContent = belege.length;
            document.getElementById('filteredBelegCount').textContent = belege.length;

            let html = '';
            
            // Group by BELEGDATUM
            const groupedByDate = {};
            let countWithDatumAllOk = 0;
            belege.forEach(beleg => {
                const date = beleg.BELEGDATUM || 'Ohne Datum';
                if (!groupedByDate[date]) {
                    groupedByDate[date] = [];
                }
                groupedByDate[date].push(beleg);

                // Count if DATUM_ALL_OK exists
                if (beleg.DATUM_ALL_OK) {
                    countWithDatumAllOk++;
                    document.getElementById('belegDatumOKCount').textContent = countWithDatumAllOk;
                }
            });
            
            // Sort dates (newest first)
            const sortedDates = Object.keys(groupedByDate).sort((a, b) => {
                if (a === 'Ohne Datum') return 1;
                if (b === 'Ohne Datum') return -1;
                return new Date(b) - new Date(a);
            });
            
            sortedDates.forEach(date => {
                const belegeForDate = groupedByDate[date];
                
                html += `<div class="date-group">`;
                html += `<div class="date-title">${date} <span class="folder-count">(${belegeForDate.length})</span></div>`;
                html += `<div class="beleg-items">`;
                
                html += `<table><thead><tr>`;
                // Get all unique keys from the first beleg (assuming all have same structure)
                const keys = Object.keys(belegeForDate[0]).filter(key => key !== 'BELEGDATUM');
                const keys1 =  Object.keys(belegeForDate[0]).filter(key => key == 'DATUM_ALL_OK');
                html += keys1.map(key =>  `<th>${key}</th>`).join('');
                html += keys.map(key => `<th>${key}</th>`).join('');
                html += `</tr></thead><tbody>`;
                
                belegeForDate.forEach(beleg => {
                    html += `<tr data-id="${beleg.ID}" onclick="loadBelegPositions(${beleg.PK_INP_BELEGE_ALL})" style="cursor:pointer;">`;
                    
                    // DATUM_ALL_OK as first column
                    html += `<td>${
                        beleg['DATUM_ALL_OK'] !== null 
                            ? `<span style="color: green;">✓ ${beleg['DATUM_ALL_OK']}</span>`
                            : '<span style="color: red;">✗</span>'
                    }</td>`;
                    
                    // PK_INP_BELEGE_ALL as second column
                    html += `<td>${beleg['PK_INP_BELEGE_ALL'] !== null ? beleg['PK_INP_BELEGE_ALL'] : ''}</td>`;
                    
                    // BEZEICHNUNG as third column
                    html += `<td>${beleg['BEZEICHNUNG'] !== null ? beleg['BEZEICHNUNG'] : ''}</td>`;
                    html += `<td>${beleg['BELEGNUMMER'] !== null ? beleg['BELEGNUMMER'] : ''}</td>`;

                    // BELEGDATUM as third column
                    html += `<td>${beleg['BELEGDATUM'] !== null ? formatDate(beleg['BELEGDATUM']) : ''}</td>`;

                    html += `<td>${beleg['BRUTTO_BETRAG'] !== null ? beleg['BRUTTO_BETRAG'] : ''}</td>`;
                    
                    // Add remaining columns (excluding the ones we've already manually added)
                    const remainingKeys = keys.filter(key => 
                        key !== 'DATUM_ALL_OK' && 
                        key !== 'PK_INP_BELEGE_ALL' && 
                        key !== 'BEZEICHNUNG' &&
                        key !== 'BRUTTO_BETRAG'
                    );
                    
                    html += remainingKeys.map(key => `<td>${beleg[key] !== null ? beleg[key] : ''}</td>`).join('');
                    
                    html += `</tr>`;
                });
                html += `</tbody></table></div></div>`;
            });
            
            document.getElementById('belegGroups').innerHTML = html;
            document.querySelector('.positions-table').innerHTML = 
                '<div class="no-selection">Wählen Sie einen Beleg aus, um die Positionen anzuzeigen</div>';
            document.querySelector('.accounts-table').innerHTML = 
                '<div class="no-selection">Wählen Sie einen Beleg aus, um die Kontobuchungen anzuzeigen</div>';
            
            // Set up click handlers for date groups
            document.querySelectorAll('.date-title').forEach(title => {
                title.addEventListener('click', function() {
                    this.classList.toggle('collapsed');
                    const content = this.nextElementSibling;
                    if (content.style.display === 'none') {
                        content.style.display = 'block';
                    } else {
                        content.style.display = 'none';
                    }
                });
            });
        }

        function loadBelegPositions(belegId) {
            currentBelegId = belegId;
            
            // Highlight selected row in all tables
            document.querySelectorAll('#belegGroups tr').forEach(row => {
                row.style.backgroundColor = '';
                if (row.getAttribute('data-id') == belegId) {
                    row.style.backgroundColor = '#d4e6f7';
                }
            });
            
            // Load positions
            fetch(`http://localhost:5000/belege_pos/${belegId}`)
                .then(response => {
                    if (!response.ok) {
                        throw new Error('Belegpositionen konnten nicht geladen werden');
                    }
                    return response.json();
                })
                .then(data => {
                    currentPositions = data;
                    displayBelegPositions(data);
                })
                .catch(error => {
                    document.querySelector('.positions-table').innerHTML = 
                        `<div class="error">Fehler beim Laden der Positionen: ${error.message}</div>`;
                });
            
            // Load accounts
            fetch(`http://localhost:5000/accounts/2/${belegId}`)
                .then(response => {
                    if (!response.ok) {
                        throw new Error('Kontobuchungen konnten nicht geladen werden');
                    }
                    return response.json();
                })
                .then(data => {
                    displayAccounts(data);
                })
                .catch(error => {
                    document.querySelector('.accounts-table').innerHTML = 
                        `<div class="error">Fehler beim Laden der Kontobuchungen: ${error.message}</div>`;
                });
        }
